import argparse
import os
from datetime import date, timedelta

import requests

# ---- Settings you might want to change -----------------------------------
KEY_ENV_VAR = "Olama"                   # name of the env variable holding your Ollama key
CLOUD_URL = "https://ollama.com/api/chat"
LOCAL_URL = "http://localhost:11434/api/chat"
CLOUD_MODEL = "gpt-oss:20b"             # default model when using Ollama cloud
LOCAL_MODEL = "llama3.2"                # default model when using local Ollama
# --------------------------------------------------------------------------

FR_API = "https://www.federalregister.gov/api/v1/documents.json"
FIELDS = ["title", "type", "publication_date", "abstract", "html_url",
          "document_number", "comments_close_on", "effective_on"]


# Step 1: fetch the rules ---------------------------------------------------
def fetch_rules(days):
    """Return Department of Education rules and proposed rules from the last `days` days."""
    start = date.today() - timedelta(days=days)
    params = {
        "conditions[agencies][]": "education-department",
        "conditions[type][]": ["RULE", "PRORULE"],
        "conditions[publication_date][gte]": start.isoformat(),
        "fields[]": FIELDS,
        "order": "newest",
        "per_page": 100,
    }
    results = []
    url = FR_API
    while url:
        resp = requests.get(url, params=params, timeout=30)
        resp.raise_for_status()
        data = resp.json()
        results.extend(data.get("results", []))
        url = data.get("next_page_url")   # already contains the query, so drop params
        params = None
    return results, start


# Step 2: organize them -----------------------------------------------------
def group_by_type(rules):
    groups = {"Rule": [], "Proposed Rule": []}
    for r in rules:
        groups.setdefault(r["type"], []).append(r)
    return groups


# Step 3: plain-English summaries with Ollama -------------------------------
def summarize(rule, model):
    """Ask Ollama for a plain-English summary. Falls back to the abstract on any error."""
    abstract = rule.get("abstract") or "(No abstract provided.)"
    api_key = os.environ.get(KEY_ENV_VAR) or os.environ.get(KEY_ENV_VAR.upper())
    url = CLOUD_URL if api_key else LOCAL_URL
    model = model or (CLOUD_MODEL if api_key else LOCAL_MODEL)
    headers = {"Authorization": f"Bearer {api_key}"} if api_key else {}

    prompt = (
        "Explain this U.S. Department of Education Federal Register document in plain "
        "English for a busy teacher or parent. Use 2 to 4 short sentences: what it "
        "changes, who it affects, and why. No jargon, no bullet points.\n\n"
        f"Title: {rule['title']}\nType: {rule['type']}\nAbstract: {abstract}"
    )
    try:
        resp = requests.post(
            url,
            headers=headers,
            json={"model": model, "stream": False,
                  "messages": [{"role": "user", "content": prompt}]},
            timeout=180,
        )
        resp.raise_for_status()
        return resp.json()["message"]["content"].strip()
    except Exception as err:
        print(f"  ! Ollama summary failed for {rule['document_number']} ({err}); using abstract.")
        return abstract


# Build the report ----------------------------------------------------------
def build_report(groups, start, use_ai, model):
    lines = [
        "# Department of Education: Federal Register rules",
        f"Published {start:%B %d, %Y} to {date.today():%B %d, %Y}",
        "",
    ]
    for rtype, heading in [("Rule", "Final rules"), ("Proposed Rule", "Proposed rules")]:
        items = groups.get(rtype, [])
        lines.append(f"## {heading} ({len(items)})")
        lines.append("")
        if not items:
            lines += ["_None in this period._", ""]
            continue
        for r in items:
            print(f"Summarizing: {r['title'][:70]}...")
            text = summarize(r, model) if use_ai else (r.get("abstract") or "(No abstract provided.)")
            lines.append(f"### {r['title']}")
            meta = f"Published {r['publication_date']}"
            if r.get("comments_close_on"):
                meta += f" · Comments due {r['comments_close_on']}"
            if r.get("effective_on"):
                meta += f" · Effective {r['effective_on']}"
            lines += [f"_{meta}_", "", text, "", f"[Read it on the Federal Register]({r['html_url']})", ""]
    return "\n".join(lines)


def main():
    parser = argparse.ArgumentParser(description="Summarize Department of Education rules from the Federal Register.")
    parser.add_argument("--days", type=int, default=30, help="how many days back to search (default 30)")
    parser.add_argument("--model", help="Ollama model to use (default depends on cloud or local)")
    parser.add_argument("--no-ai", action="store_true", help="skip Ollama and use the official abstracts")
    parser.add_argument("--out", default="education_rules_report.md", help="where to save the report")
    args = parser.parse_args()

    rules, start = fetch_rules(args.days)
    print(f"Found {len(rules)} Department of Education rules since {start}.\n")
    report = build_report(group_by_type(rules), start, not args.no_ai, args.model)

    with open(args.out, "w", encoding="utf-8") as f:
        f.write(report)
    print("\n" + report)
    print(f"\nSaved to {args.out}")


if __name__ == "__main__":
    main()
